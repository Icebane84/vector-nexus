--- SCRIPT: res://scripts/autoloads/SaveManager.gd ---
```gdscript
extends Node

const SAVE_PATH = "user://savegame.save"

# Signals to notify the rest of the game when I/O operations finish
signal game_saved
signal game_loaded

## Gathers data from all nodes in the "Persist" group and writes to disk.
func save_game() -> void:
	var save_file = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if save_file == null:
		printerr("SaveManager: Failed to open save file for writing.")
		return

	var save_nodes = get_tree().get_nodes_in_group("Persist")
	for node in save_nodes:
		if not node.has_method("save"):
			printerr("SaveManager: Node '%s' in Persist group lacks save() method." % node.name)
			continue

		var node_data = node.call("save")
		var json_string = JSON.stringify(node_data)
		save_file.store_line(json_string)

	print("SaveManager: Game saved successfully.")
	game_saved.emit()

## Reads the save file and routes the data back to the appropriate systems.
func load_game() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		print("SaveManager: No save file found. Starting fresh.")
		return

	var save_file = FileAccess.open(SAVE_PATH, FileAccess.READ)
	if save_file == null:
		printerr("SaveManager: Failed to open save file for reading.")
		return

	while save_file.get_position() < save_file.get_length():
		var json_string = save_file.get_line()
		var json = JSON.new()
		var parse_result = json.parse(json_string)
		if parse_result != OK:
			printerr("SaveManager: JSON Parse Error: ", json.get_error_message())
			continue

		var node_data = json.get_data()

		# --- DATA ROUTING LOGIC ---
		if node_data.has("scene_path"):
			var scene_path: String = node_data["scene_path"]
			if not ResourceLoader.exists(scene_path):
				printerr("SaveManager: Scene path not found: ", scene_path)
				continue
			var new_instance: Node = load(scene_path).instantiate()
			get_tree().current_scene.add_child(new_instance)
			if new_instance.has_method("load_data"):
				new_instance.call("load_data", node_data)
			else:
				printerr("SaveManager: New instance lacks load_data() method.")
		elif node_data.has("node_path"):
			var node_path: NodePath = node_data["node_path"]
			var node: Node = get_node_or_null(node_path)
			if node and node.has_method("load_data"):
				node.call("load_data", node_data)
			else:
				printerr(
					"SaveManager: Could not find node or load_data method for path: ",
					node_path
				)
		else:
			printerr("SaveManager: Save data entry has no 'scene_path' or 'node_path' to route it.")

	print("SaveManager: Game loaded successfully.")
	game_loaded.emit()

```

```gdscript
# [GVRN]
# Artifact ID:   COMP.Combat.PooledProjectile
# Description:   Zero-Allocation projectile logic.
# Author:        Architect

extends Area3D
class_name PooledProjectile

signal returned_to_pool(proj: PooledProjectile)

@export var speed: float = 20.0
@export var max_lifetime: float = 3.0
@export var hitbox: HitboxComponent

var _active: bool = false
var _lifetime_timer: float = 0.0
var _direction: Vector3 = Vector3.FORWARD

func _ready() -> void:
	if hitbox and not hitbox.area_entered.is_connected(_on_hitbox_entered):
		hitbox.area_entered.connect(_on_hitbox_entered)

func fire(start_pos: Vector3, dir: Vector3) -> void:
	global_position = start_pos
	_direction = dir.normalized()
	
	# Align projectile to face target direction
	if _direction != Vector3.ZERO and _direction.cross(Vector3.UP).length() > 0.01:
		look_at(global_position + _direction, Vector3.UP)
		
	_lifetime_timer = 0.0
	_active = true
	show()

func _physics_process(delta: float) -> void:
	if not _active: return
	
	# Manual timer avoids SceneTreeTimer garbage collection spikes
	_lifetime_timer += delta
	if _lifetime_timer >= max_lifetime:
		return_to_pool()
		return
		
	global_position += _direction * speed * delta

func _on_hitbox_entered(_area: Area3D) -> void:
	return_to_pool()

func return_to_pool() -> void:
	_active = false
	hide()
	returned_to_pool.emit(self)
```

```gdscript
# [GVRN]
# Artifact ID: COMP.Camera.LockOn
# Description: Rotational binding logic for third-person camera targeting.
# Author: Architect

extends Node3D

class_name LockOnComponent

signal target_changed(target: Node3D)

@export var max_lock_distance: float = 30.0
@export var _mock_targets: Array[Node] = []

var _current_target: Node3D = null
var current_target: Node3D:
	get: return _current_target
	set(v):
		if _current_target != v:
			_current_target = v
			# Signal locally for portability
			target_changed.emit(v)

func find_closest_target(camera: Camera3D) -> Node3D:
	var best_target: Node3D = null
	var min_dist_to_center: float = INF
	var viewport_center = camera.get_viewport().get_visible_rect().size / 2.0
	
	var targets = _mock_targets if _mock_targets.size() > 0 \
		else get_tree().get_nodes_in_group("enemy")
	
	for t in targets:
		if not is_instance_valid(t) or not t is Node3D: continue
		if global_position.distance_to(t.global_position) > max_lock_distance: continue
		if camera.is_position_behind(t.global_position): continue
			
		var screen_pos = camera.unproject_position(t.global_position)
		var center_dist = screen_pos.distance_to(viewport_center)
		
		if center_dist < min_dist_to_center:
			min_dist_to_center = center_dist
			best_target = t
			
	return best_target

func find_target_in_direction(camera: Camera3D, flick_dir: Vector2) -> Node3D:
	if not is_instance_valid(current_target):
		return find_closest_target(camera)
		
	var best_target: Node3D = null
	var min_score: float = INF
	var current_screen_pos = camera.unproject_position(current_target.global_position)
	var targets = _mock_targets if _mock_targets.size() > 0 else get_tree().get_nodes_in_group("enemy")
	
	for t in targets:
		if t == current_target or not is_instance_valid(t) or not t is Node3D: continue
		if global_position.distance_to(t.global_position) > max_lock_distance: continue
		if camera.is_position_behind(t.global_position): continue
			
		var screen_pos = camera.unproject_position(t.global_position)
		var dir_to_candidate = (screen_pos - current_screen_pos).normalized()
		
		# Ensure candidate is in the general direction of the flick (dot > 0.2)
		var dot = dir_to_candidate.dot(flick_dir.normalized())
		if dot > 0.2:
			var center_dist = screen_pos.distance_to(current_screen_pos)
			var score = center_dist * (1.0 - dot + 0.1) 
			
			if score < min_score:
				min_score = score
				best_target = t
				
	return best_target

```

```gdscript
# [GVRN]
# Artifact ID: COMP.Physics.Health
# Description: Core health management and invincibility handling.
# Author: Architect

extends Node
class_name HealthComponent

signal health_changed(current: float, maximum: float)
signal damaged(amount: float, source: Node)
signal healed(amount: float)
signal died

@export_group("Stats")
@export var max_health: float = 100.0
@export var invincibility_time: float = 0.5

var current_health: float:
	set(value):
		var old := current_health
		current_health = clamp(value, 0.0, max_health)
		if current_health != old:
			health_changed.emit(current_health, max_health)

var _invincible: bool = false

func _ready() -> void:
	current_health = max_health

func receive_damage(amount: float, source: Node = null) -> float:
	if _invincible or current_health <= 0:
		return 0.0

	var actual := minf(amount, current_health)
	current_health -= actual
	damaged.emit(actual, source)

	if current_health <= 0:
		died.emit()
	elif invincibility_time > 0:
		_start_invincibility()

	return actual

func heal(amount: float) -> float:
	var actual := minf(amount, max_health - current_health)
	current_health += actual
	if actual > 0:
		healed.emit(actual)
	return actual

func _start_invincibility() -> void:
	_invincible = true
	# PHOENIX-FIX: Ensuring timer is independent of time scale during hit-stops
	await get_tree().create_timer(invincibility_time, true, false, true).timeout
	_invincible = false

```

```gdscript
# [GVRN]
# Artifact ID: COMP.Physics.Hurtbox
# Description: Centralized damage reception logic for living entities.
# Author: Architect

extends Area3D

class_name HurtboxComponent

signal hit_received(attack: AttackData)
signal parry_successful(attacker_pos: Vector3)

@export var health_component: HealthComponent
@export var poise_component: PoiseComponent
@export var team_id: int = 0

# SKILL-001: Backing-Field Resilience
var _is_invincible: bool = false
@export var is_invincible: bool:
	get: return _is_invincible
	set(v): _is_invincible = v

var _is_parry_window: bool = false
@export var is_parry_window: bool:
	get: return _is_parry_window
	set(v): _is_parry_window = v

func receive_damage(attack: AttackData) -> void:
	# 1. Team Filter
	if attack.team_id == team_id: 
		return

	# 2. I-frame Check (User Requested logic)
	if _is_invincible:
		print("PHOENIX_LOG: Hit negated by I-frames.")
		return

	# 3. Parry Logic
	if _is_parry_window and attack.can_be_parried:
		print("PHOENIX_LOG: Parry successful!")
		parry_successful.emit(attack.attacker_pos)
		_trigger_parry_feedback()
		return

	# 4. Standard Damage
	if health_component:
		health_component.receive_damage(attack.damage)
	
	if poise_component:
		poise_component.apply_poise_damage(attack.poise_damage)
	
	hit_received.emit(attack)

func _trigger_parry_feedback() -> void:
	# Guard Director calls for headless tests
	if Director.instance.active_combat_system:
		Director.instance.active_combat_system.trigger_parry_hitstop()
```

```gdscript
"""
[GVRN]
Artifact ID:   COMP.State.Dodge
Description:   Handles directional dodge rolling with state-locking and duration limits.
"""
extends Node

@export_group("Dodge Settings")
@export var dodge_speed: float = 18.0
@export var dodge_duration: float = 0.4

var _dodge_timer: float = 0.0
var _dodge_direction: Vector3 = Vector3.ZERO

func enter_state(entity: CharacterBody3D, direction_input: Vector3) -> void:
	_dodge_timer = dodge_duration
	
	# DAMP: Explicit fallback if the player dodges without pressing a movement key
	if direction_input == Vector3.ZERO:
		var cam_basis := entity.global_transform.basis
		_dodge_direction = Vector3(-cam_basis.z.x, 0.0, -cam_basis.z.z).normalized()
	else:
		_dodge_direction = direction_input.normalized()
		
	# Snap entity rotation to face the dodge direction immediately
	if _dodge_direction != Vector3.ZERO:
		var target_angle := atan2(-_dodge_direction.x, -_dodge_direction.z)
		entity.rotation.y = target_angle

func update_physics(entity: CharacterBody3D, delta: float) -> String:
	_dodge_timer -= delta
	
	# Lock velocity to the dodge direction
	entity.velocity.x = _dodge_direction.x * dodge_speed
	entity.velocity.z = _dodge_direction.z * dodge_speed
	
	# Apply gravity just in case we dodge off a ledge
	if not entity.is_on_floor():
		var gravity: float = ProjectSettings.get_setting("physics/3d/default_gravity")
		entity.velocity.y -= gravity * delta
		
	entity.move_and_slide()
	
	# Explicit exit once duration expires
	if _dodge_timer <= 0.0:
		return "Idle"
		
	return ""

```

```gdscript
		_regen_timer -= delta
		return

	if _current_stamina <= 0:
		stamina_depleted.emit()
	
	return true

	return _current_stamina >= _max_stamina
```

```gdscript
[GVRN]
Artifact ID: InputBufferComponent
Description: Centralized input buffering system.
"""

extends Node
class_name InputBufferComponent

# How long an input is remembered (in seconds)
@export var buffer_window: float = 0.2 

# Stores the action names and their remaining buffer time
var _buffered_actions: Dictionary = {}

func _process(delta: float) -> void:
	# Degrade the timers for all buffered actions
	var actions_to_remove = []
	for action in _buffered_actions.keys():
		_buffered_actions[action] -= delta
		if _buffered_actions[action] <= 0.0:
			actions_to_remove.append(action)
			
	# Clean up expired actions
	for action in actions_to_remove:
		_buffered_actions.erase(action)

## Call this from your main input handling script when a button is pressed
func buffer_action(action: String) -> void:
	_buffered_actions[action] = buffer_window

## Call this from your states to check if an action is queued up
func is_action_buffered(action: String) -> bool:
	return _buffered_actions.has(action)

## Call this when a state transitions to consume the input so it doesn't fire twice
func consume_action(action: String) -> void:
	if _buffered_actions.has(action):
		_buffered_actions.erase(action)

## Optional: clear the whole buffer (useful when the player gets stunned/hit)
func clear_buffer() -> void:
	_buffered_actions.clear()
```

```gdscript
# [GVRN]
# Artifact ID: FABRIC.Logic.StateMachine
# Description: Standard hierarchical state machine pattern.
# Version: [SOVEREIGN]
# Author: Architect

extends Node

class_name StateMachine

signal state_changed(from_state: StringName, to_state: StringName)

@export var initial_state: State

var current_state: State
var states: Dictionary = {}

func _ready() -> void:
	# PHOENIX-GVRN: Automatic Recursive Registration
	_register_states(self)

	# Delay start to ensure actor/animation pointers are valid in parent
	await get_tree().process_frame

	if initial_state:
		current_state = initial_state
		current_state.process_mode = Node.PROCESS_MODE_INHERIT
		current_state.enter()

func _register_states(parent: Node) -> void:
	for child in parent.get_children():
		if child is State:
			states[child.name] = child
			child.state_machine = self
			child.process_mode = Node.PROCESS_MODE_DISABLED
			# Recurse for nested states (HSM Support)
			_register_states(child)

func _process(delta: float) -> void:
	if current_state:
		current_state.update(delta)

func _physics_process(delta: float) -> void:
	if current_state:
		current_state.physics_update(delta)

func _unhandled_input(event: InputEvent) -> void:
	if current_state:
		current_state.handle_input(event)

func transition_to(state_name: StringName, msg: Dictionary = {}) -> void:
	if not states.has(state_name):
		# PHOENIX-LOG: Resilient Error Handling
		push_error("State '%s' not found in %s" % [state_name, get_parent().name])
		return

	var previous_state := current_state
	if previous_state:
		previous_state.exit()
		previous_state.process_mode = Node.PROCESS_MODE_DISABLED

	current_state = states[state_name]
	current_state.process_mode = Node.PROCESS_MODE_INHERIT
	current_state.enter(msg)

	state_changed.emit(previous_state.name if previous_state else &"", current_state.name)

```

```gdscript
	if current_state != null:
		current_state.enter()
```

```gdscript
	if not actor.get("target"):
		_find_new_target()
		return

	var target: Node3D = actor.target
	var dist: float = actor.global_position.distance_to(target.global_position)
	
	# Transition Logic using Enums (Internal logic maps to state_machine strings)

		return

	# Movement Execution
	_execute_movement(target.global_position)

func _find_new_target() -> void:
	var players = get_tree().get_nodes_in_group("Player")
	if players.size() > 0:
		actor.target = players[0]
	else:
		state_machine.transition_to("AIIdleState")

func _check_transitions(dist: float, target: Node3D) -> bool:
	if dist < attack_range:
		state_machine.transition_to(&"AIAttackState") 
		return true
		
	# Line of Sight & Interest Check
	if dist > lose_interest_range or not _has_line_of_sight(target):
		actor.target = null
		state_machine.transition_to(&"AIIdleState") 
		return true
	
	return false

func _has_line_of_sight(target: Node3D) -> bool:
	if not los_ray: return true # Fallback if raycast is missing
	los_ray.look_at(target.global_position + Vector3.UP)
	return los_ray.is_colliding() and los_ray.get_collider() == target

func _execute_movement(target_pos: Vector3) -> void:
	var direction: Vector3 = (target_pos - actor.global_position).normalized()
	actor.velocity = direction * move_speed
	actor.move_and_slide()
```

```gdscript
			lock_on.current_target = lock_on.find_closest_target(_camera)

	# Handle camera rotation and target flicking
	if event is InputEventMouseMotion and Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
		# If locked on, handle flick-to-switch
		if lock_on and is_instance_valid(lock_on.current_target):
			var flick_threshold: float = 40.0
			if abs(event.relative.x) > flick_threshold:
				var flick_dir = Vector2(sign(event.relative.x), 0)
				var new_target = lock_on.find_target_in_direction(_camera, flick_dir)
				if new_target:
					lock_on.current_target = new_target
		else:
			# Free-look
			rotation.y -= event.relative.x * mouse_sensitivity
			rotation.x -= event.relative.y * mouse_sensitivity
			rotation.x = clamp(rotation.x, min_pitch, max_pitch)
		
	# Allow escaping the mouse lock for testing purposes
	if event.is_action_pressed("ui_cancel"):
		if Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
			Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
		else:
			Input.mouse_mode = Input.MOUSE_MODE_CAPTURED

func _physics_process(_delta: float) -> void:
	# Strictly follow the assigned target, maintaining the vertical offset
	if target:
		global_position = target.global_position + Vector3(0, vertical_offset, 0)

	# Smoothly rotate the SpringArm to face the locked-on target
	if lock_on and is_instance_valid(lock_on.current_target):
		var target_pos: Vector3 = lock_on.current_target.global_position
		var dir: Vector3 = (target_pos - global_position).normalized()
		
		rotation.y = lerp_angle(rotation.y, atan2(-dir.x, -dir.z), 10.0 * _delta)
		rotation.x = lerp_angle(rotation.x, asin(dir.y), 10.0 * _delta)
```

```gdscript
# object_pool.gd
class_name ObjectPool
extends Node

## PHOENIX ARCHITECT: SKILL-GAM-001 ALIGNMENT
## ANTI_PATTERN_ZERO: Prevents object creation in hot loops via pooling.

@export var pooled_scene: PackedScene
@export var initial_size: int = 20
@export var can_grow: bool = true

var _available: Array[Node] = []
var _in_use: Array[Node] = []

func _ready() -> void:
    if not pooled_scene:
        push_error("ObjectPool Error: pooled_scene must be assigned in the Inspector.")
        return
    _initialize_pool()

func _initialize_pool() -> void:
    for i in initial_size:
        _create_instance()

func _create_instance() -> Node:
    var instance := pooled_scene.instantiate()
    instance.process_mode = Node.PROCESS_MODE_DISABLED
    instance.visible = false
    add_child(instance)
    _available.append(instance)

    if instance.has_signal("returned_to_pool"):
        instance.returned_to_pool.connect(_return_to_pool.bind(instance))

    return instance

func get_instance() -> Node:
    var instance: Node

    if _available.is_empty():
        if can_grow:
            instance = _create_instance()
            _available.erase(instance)
        else:
            push_warning("Pool exhausted and cannot grow")
            return null
    else:
        instance = _available.pop_back()

    instance.process_mode = Node.PROCESS_MODE_INHERIT
    instance.visible = true
    _in_use.append(instance)

    if instance.has_method("on_spawn"):
        instance.on_spawn()

    return instance

func _return_to_pool(instance: Node) -> void:
    if not instance in _in_use:
        return

    _in_use.erase(instance)

    if instance.has_method("on_despawn"):
        instance.on_despawn()

    instance.process_mode = Node.PROCESS_MODE_DISABLED
    instance.visible = false
    _available.append(instance)

```

```gdscript
func _register_states(parent: Node) -> void:
	for child in parent.get_children():
		if child is PatternState:
			states[child.name] = child
			child.state_machine = self
			child.process_mode = Node.PROCESS_MODE_DISABLED
			_register_states(child)

		current_state.update(delta)

	if current_state:
		current_state.physics_update(delta)

func _unhandled_input(event: InputEvent) -> void:

		current_state.handle_input(event)

func transition_to(state_name: StringName, msg: Dictionary = {}) -> void:

	if previous_state:
		previous_state.exit()
		previous_state.process_mode = Node.PROCESS_MODE_DISABLED

	current_state = states[state_name]
	current_state.process_mode = Node.PROCESS_MODE_INHERIT
	current_state.enter(msg)
```

```gdscript
# The maximum mana a character can have.
@export var max_mana: float = 100.0

# A signal to notify other nodes (like a UI bar) when mana has changed.
signal mana_changed(new_mana_value, max_mana_value)

# The private "backing field". It stores the true value of mana.
# The underscore prefix is a GDScript convention for private variables.
var _current_mana: float

# The public property that other scripts will interact with.
# We define custom 'get' and 'set' logic for it.
var current_mana: float:
	# The 'get' function simply returns the current value of our backing field.
	get:
		return _current_mana

	# The 'set' function contains the core logic for the pattern.
	# It runs every time you write a new value to 'current_mana'.
	set(new_value):
		# First, we clamp the incoming value to ensure it's within a valid range.
		var clamped_value = clamp(new_value, 0.0, max_mana)

		# IMPORTANT: We check if the new, clamped value is different from the old one.
		# This prevents the 'mana_changed' signal from firing unnecessarily,
		# which avoids wasted processing and potential bugs.
		if clamped_value != _current_mana:
			_current_mana = clamped_value
			mana_changed.emit(_current_mana, max_mana)
			# SKILL-004: Route the event through the global synapse
			GameEvents.instance.player_mana_changed.emit(_current_mana, max_mana)

	# This will automatically run the 'set' logic.
	current_mana = max_mana

	if current_mana >= mana_cost:
		# Subtracting from the property also triggers the 'set' logic,
		# ensuring the value is updated and the signal is emitted correctly.
		current_mana -= mana_cost
		print("Spell cast! Mana is now: ", current_mana)
```

```gdscript
# Description: The primary Head-Up Display controller.
#              Orchestrates health bars, lock-on indicators, and status icons.
# Version:       1.1 [SOVEREIGN]

	Director.instance.player_health_ready.connect(_on_player_health_ready)
	GameEvents.instance.lock_on_target_changed.connect(_on_lock_on_target_changed)
	GameEvents.instance.combat_state_changed.connect(_on_combat_state_changed)
	# Wait for Stamina component to be registered to the Director
	# This avoids race conditions during player instantiation
	_setup_stamina()

func _process(_delta: float) -> void:
	if not _locked_target or not is_instance_valid(_locked_target):
		if %LockOnReticle.visible:
			%LockOnReticle.hide()
		return
	
	var cam := get_viewport().get_camera_3d()
	if not cam: return
	
	# SKILL-006: Surface-Aware (Targeting the specific marker)
	var marker = _locked_target.get_node_or_null("LockOnMarker")
	var target_pos = marker.global_position if marker else _locked_target.global_position
	
	if cam.is_position_behind(target_pos):
		%LockOnReticle.hide()
	else:
		%LockOnReticle.show()
		%LockOnReticle.position = cam.unproject_position(target_pos)

func _on_lock_on_target_changed(target: Node3D) -> void:

	%LockOnReticle.visible = (target != null)

	await get_tree().process_frame
	var stamina = Director.instance.player_stamina_component
	if stamina:
		if not stamina.stamina_changed.is_connected(_on_stamina_changed):
			stamina.stamina_changed.connect(_on_stamina_changed)
		_on_stamina_changed(stamina.current_stamina, stamina.max_stamina)

func _on_player_health_ready(node: HealthComponent) -> void:
	if not node.health_changed.is_connected(_on_health_changed):
		node.health_changed.connect(_on_health_changed)
	_on_health_changed(node.current_health, node.max_health)

func _on_health_changed(curr: float, max_v: float) -> void:
	%HealthBar.max_value = max_v
	%HealthBar.value = curr

func _on_stamina_changed(curr: float, max_v: float) -> void:
	%StaminaBar.max_value = max_v
	%StaminaBar.value = curr
	
	# PHOENIX-PREF: Hidden when full
	%StaminaBar.visible = curr < max_v

func _on_combat_state_changed(actor: Node3D, new_state: T.CombatState) -> void:
	if new_state in [T.CombatState.CHASE, T.CombatState.ATTACK]:
		_active_combatants[actor] = true
	else:
		_active_combatants.erase(actor)
		
	# Cleanup invalid/freed actors (SKILL-002 Memory Safety)
	for key in _active_combatants.keys():
		if not is_instance_valid(key):
			_active_combatants.erase(key)
			
	var indicator := get_node_or_null("%CombatIndicator") as Control
	if indicator:
		indicator.visible = not _active_combatants.is_empty()
```

```gdscript
# [GVRN]
# Artifact ID: UI.TransitionScreen
# Description: Global screen fader for state transitions.
extends CanvasLayer
class_name TransitionScreen

@export var fade_rect: ColorRect
@export var fade_duration: float = 1.0

func _ready() -> void:

func fade_to_black() -> Signal:
	fade_rect.mouse_filter = Control.MOUSE_FILTER_STOP # Block clicks while fading
	var tween = create_tween()
	tween.tween_property(fade_rect, "modulate:a", 1.0, fade_duration)
	return tween.finished

func fade_to_clear() -> Signal:
	var tween = create_tween()
	tween.tween_property(fade_rect, "modulate:a", 0.0, fade_duration)
	tween.finished.connect(func(): fade_rect.mouse_filter = Control.MOUSE_FILTER_IGNORE)
	return tween.finished
```
